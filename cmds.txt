//restricciones a los usuarios

net user <usuario> /expires:never /domain                       //expira
net user <usuario> /time:Dia,Hora-Hasta,Dia,Hora-Hasta /domain  // limitar horas de inicio de sesion puedes concatenar tantas dia hora como quieras
net user <usuario> /time:all /domain                            //deslimita las horas de inicio de sesion
